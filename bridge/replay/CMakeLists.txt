cmake_minimum_required(VERSION 3.20)
project(renderdocx_replay LANGUAGES CXX)

set(RENDERDOC_LIBRARY "" CACHE FILEPATH "Import library or shared library built from this checkout")
set(RENDERDOC_CORE_COMMIT "" CACHE STRING "Exact git commit reported by the matching RenderDoc core")
if(NOT EXISTS "${RENDERDOC_LIBRARY}")
  message(FATAL_ERROR "Set RENDERDOC_LIBRARY to the RenderDoc core built from this checkout")
endif()
if(NOT RENDERDOC_CORE_COMMIT MATCHES "^[0-9a-fA-F]+$")
  message(FATAL_ERROR "Set RENDERDOC_CORE_COMMIT to the matching core's full git commit")
endif()
string(LENGTH "${RENDERDOC_CORE_COMMIT}" COMMIT_LENGTH)
if(NOT COMMIT_LENGTH EQUAL 40)
  message(FATAL_ERROR "RENDERDOC_CORE_COMMIT must have 40 hexadecimal characters")
endif()

add_executable(renderdocx-replay main.cpp)
add_executable(renderdocx-target target.cpp)
foreach(target renderdocx-replay renderdocx-target)
target_compile_features(${target} PRIVATE cxx_std_17)
target_compile_definitions(${target} PRIVATE RENDERDOCX_CORE_COMMIT="${RENDERDOC_CORE_COMMIT}")
target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../renderdoc")
target_link_libraries(${target} PRIVATE "${RENDERDOC_LIBRARY}")
if(WIN32)
  target_compile_definitions(${target} PRIVATE RENDERDOC_PLATFORM_WIN32)
elseif(APPLE)
  target_compile_definitions(${target} PRIVATE RENDERDOC_PLATFORM_APPLE)
elseif(UNIX)
  target_compile_definitions(${target} PRIVATE RENDERDOC_PLATFORM_LINUX)
endif()
if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /utf-8)
endif()
endforeach()
